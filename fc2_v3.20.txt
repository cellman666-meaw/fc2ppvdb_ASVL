    const cleanAlnum = (s) => s.toLowerCase().replace(RE_NON_ALNUM, '');
    const numericOnly = (s) => s.replace(RE_NON_NUM, '') || s;

    function resolveAbsoluteUrl(href, domain) {
        if (!href) return '';
        if (href.charCodeAt(0) === 104 && href.startsWith('http')) return href; // h
        return `https://${domain}/${href.charCodeAt(0) === 47 ? href.slice(1) : href}`; // /
    }

    // Optimized findDirectPostLink: single pass over <a>, no Array.from, early exits
    function findDirectPostLink(doc, domain, id) {
        const cleanId = cleanAlnum(id);
        if (!cleanId) return null;
        // Fast path: use doc.links (live HTMLCollection) if available
        const links = doc.links || doc.querySelectorAll('a');
        // Avoid scanning huge pages fully - cap at 600 links (cards pages rarely exceed)
        const len = Math.min(links.length, 600);
        for (let i = 0; i < len; i++) {
            const a = links[i];
            const hrefRaw = a.getAttribute('href');
            if (!hrefRaw) continue;
            const href = hrefRaw.toLowerCase();
            if (href === '#' || href === '/' || href.charCodeAt(0) === 106) continue; // j for javascript:
            if (href.includes('?s=') || href.includes('keyword=') || href.includes('search') ||
                href.includes('/category/') || href.includes('/tag/') || href.includes('/page/') || href.includes('/genre/')) continue;

            // cheapest check first: href
            const cleanHref = href.replace(RE_NON_ALNUM, '');
            if (cleanHref.includes(cleanId)) {
                if (href.includes(domain) || hrefRaw.charCodeAt(0) === 47 || hrefRaw.charCodeAt(0) === 46) return a;
            } else {
                // only compute text/alt if href miss
                const cleanText = a.textContent.toLowerCase().replace(RE_NON_ALNUM, '');
                if (cleanText.includes(cleanId)) {
                    if (href.includes(domain) || hrefRaw.charCodeAt(0) === 47 || hrefRaw.charCodeAt(0) === 46) return a;
                    continue;
                }
                const img = a.querySelector('img');
                if (img) {
                    const alt = img.getAttribute('alt');
                    if (alt && alt.toLowerCase().replace(RE_NON_ALNUM, '').includes(cleanId)) {
                        if (href.includes(domain) || hrefRaw.charCodeAt(0) === 47 || hrefRaw.charCodeAt(0) === 46) return a;
                    }
                }
            }
        }
        return null;
    }

    function acquireGlobalSlot() {
        if (globalActive < GLOBAL_MAX_CONCURRENT) { globalActive++; return true; }
        return false;
    }
    function releaseGlobalSlot() {
        globalActive = Math.max(0, globalActive - 1);
        if (globalWaitQueue.length) {
            const fn = globalWaitQueue.shift();
            globalActive++;
            fn();
        }
    }
    function enqueueGlobal(fn) {
        if (acquireGlobalSlot()) fn();
        else globalWaitQueue.push(fn);
    }

    class FloatingMenu {
        constructor(titleText) {
            this.isDragging = false;
            this.wasDragged = false;
            this.offsetX = 0;
            this.offsetY = 0;
            this.rafId = 0;
            this.nextX = 0; this.nextY = 0;
            this.createMenu(titleText);
        }
        createMenu(titleText) {
            this.menu = document.createElement("div");
            this.menu.id = "floating-menu";
            document.body.appendChild(this.menu);
            this.title = document.createElement("div");
            this.title.id = "floating-menu-title";
            this.title.textContent = `🔍 ${titleText}`;
            this.menu.appendChild(this.title);
            this.content = document.createElement("div");
            this.content.id = "floating-menu-content";
            this.content.style.display = "none";
            this.menu.appendChild(this.content);
            this._bindEvents();
        }
        _bindEvents() {
            this.title.addEventListener("click", () => {
                if (this.wasDragged) { this.wasDragged = false; return; }
                this.content.style.display = this.content.style.display === "none" ? "flex" : "none";
            });
            this.title.addEventListener("pointerdown", (e) => {
                this.isDragging = true;
                this.wasDragged = false;
                const r = this.menu.getBoundingClientRect();
                this.offsetX = e.clientX - r.left;
                this.offsetY = e.clientY - r.top;
                this.title.setPointerCapture(e.pointerId);
                this.title.style.cursor = "grabbing";
                document.body.style.userSelect = "none";
                e.preventDefault();
            });
            const onMove = (e) => {
                if (!this.isDragging) return;
                this.wasDragged = true;
                this.nextX = e.clientX - this.offsetX;
                this.nextY = e.clientY - this.offsetY;
                if (!this.rafId) {
                    this.rafId = requestAnimationFrame(() => {
                        this.rafId = 0;
                        this.menu.style.left = `${this.nextX}px`;
                        this.menu.style.top = `${this.nextY}px`;
                        this.menu.style.transform = "none";
                    });
                }
            };
            const onUp = () => {
                if (!this.isDragging) return;
                this.isDragging = false;
                this.title.style.cursor = "grab";
                document.body.style.userSelect = "";
                if (this.rafId) { cancelAnimationFrame(this.rafId); this.rafId = 0; }
                // reset wasDragged after click suppression window
                if (this.wasDragged) setTimeout(() => { this.wasDragged = false; }, 120);
            };
            document.addEventListener("pointermove", onMove, { passive: true });
            document.addEventListener("pointerup", onUp);
            document.addEventListener("pointercancel", onUp);
        }
        addContent(el) { this.content.appendChild(el); }
    }

    // ─────────────────────────────────────────────
    // Utils - LRU session cache + batched GM storage
    // ─────────────────────────────────────────────
    class Utils {
        static sessionCache = new Map(); // LRU
        static MAX_SESSION = 600;
        static pendingSearches = new Map();

        static _touchLRU(key) {
            const v = this.sessionCache.get(key);
            if (v !== undefined) { this.sessionCache.delete(key); this.sessionCache.set(key, v); }
        }
        static _setLRU(key, val) {
            if (this.sessionCache.has(key)) this.sessionCache.delete(key);
            else if (this.sessionCache.size >= this.MAX_SESSION) {
                const first = this.sessionCache.keys().next().value;
                this.sessionCache.delete(first);
            }
            this.sessionCache.set(key, val);
        }

        static showLoading(container, siteName) {
            if (!container) return null;
            const el = document.createElement("span");
            el.textContent = `🔍 [${siteName}]...`;
            el.className = "fc2cmadb-loading";
            el.dataset.site = siteName;
            container.appendChild(el);
            return el;
        }
        static removeLoading(el) { if (el) el.remove(); }

        static createButton(url, text, bgColorClass) {
            const a = document.createElement("a");
            a.href = url;
            a.textContent = text;
            a.className = "fc2cmadb-search-btn";
            a.style.backgroundColor = COLOR_MAP[bgColorClass] || '#6366f1';
            a.target = "_blank";
            a.rel = "noopener noreferrer";
            return a;
        }

        static async cacheResult(key, data, expiryHours = 12) {
            this._setLRU(key, data);
            const expiry = Date.now() + expiryHours * 3_600_000;
            // fire-and-forget GM.setValue to not block queue (still await for correctness, but not idle)
            try { await GM.setValue(key, { data, expiry }); } catch {}
        }

        static async getCachedResult(key) {
            if (this.sessionCache.has(key)) {
                const v = this.sessionCache.get(key);
                this._touchLRU(key);
                return v;
            }
            let cached = null;
            try { cached = await GM.getValue(key, null); } catch {}
            if (!cached) { this._setLRU(key, null); return null; }
            if (cached.expiry < Date.now()) {
                try { await GM.setValue(key, null); } catch {}
                this._setLRU(key, null);
                return null;
            }
            this._setLRU(key, cached.data);
            return cached.data;
        }
    }

    // ─────────────────────────────────────────────
    // Site - global throttle + batch dedup
    // ─────────────────────────────────────────────
    class Site {
        constructor(name, searchUrlFunc, responseHandler, buttonColor, queryInterval = 800, maxConcurrentSearches = 2) {
            this.name = name;
            this.searchUrlFunc = searchUrlFunc;
            this.responseHandler = responseHandler;
            this.buttonColor = buttonColor;
            this.queryInterval = queryInterval;
            this.maxConcurrentSearches = maxConcurrentSearches;
            this.queue = [];
            this.active = 0;
        }
        addToQueue(id, container) {
            this.queue.push({ id, container });
            this._schedule();
        }
        _schedule() {
            while (this.queue.length && this.active < this.maxConcurrentSearches) {
                const task = this.queue.shift();
                this.active++;
                const run = async () => {
                    try { await this.search(task.id, task.container); }
                    catch (e) { console.error(`[${this.name}]`, e); }
                    finally {
                        this.active--;
                        releaseGlobalSlot();
                        // per-site interval before next
                        setTimeout(() => this._schedule(), this.queryInterval);
                    }
                };
                enqueueGlobal(run);
                // if queued globally, active already counted, but global slot may be pending
                if (globalActive > GLOBAL_MAX_CONCURRENT) break;
            }
        }
        async search(id, container) {
            if (!container.isConnected) return;
            const cacheKey = `${this.name}_${id}`;
            // check pending first (dedup identical id across multiple cards)
            if (Utils.pendingSearches.has(cacheKey)) {
                const url = await Utils.pendingSearches.get(cacheKey);
                if (url && container.isConnected) this.addButton(container, url);
                return;
            }
            const cached = await Utils.getCachedResult(cacheKey);
            if (cached) {
                if (cached.url && container.isConnected) this.addButton(container, cached.url);
                return;
            }
            const searchUrl = this.searchUrlFunc(id);
            const loadingEl = Utils.showLoading(container, this.name);

            const promise = new Promise((resolve) => {
                GM_xmlhttpRequest({
                    method: "GET",
                    url: searchUrl,
                    timeout: 12000,
                    headers: { "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8" },
                    onload: (res) => {
                        if (res.status !== 200) {
                            Utils.removeLoading(loadingEl);
                            Utils.cacheResult(cacheKey, { url: '' }, 0.05);
                            resolve('');
                            return;
                        }
                        const previewData = { url: '' };
                        try { this.responseHandler(res, id, previewData); } catch (e) { console.error(`[${this.name}] parse`, e); }
                        if (previewData.url && container.isConnected) this.addButton(container, previewData.url);
                        Utils.removeLoading(loadingEl);
                        Utils.cacheResult(cacheKey, previewData, previewData.url ? 24 : 12);
                        resolve(previewData.url);
                    },
                    onerror: () => { Utils.removeLoading(loadingEl); Utils.cacheResult(cacheKey, { url: '' }, 0.05); resolve(''); },
                    ontimeout: () => { Utils.removeLoading(loadingEl); Utils.cacheResult(cacheKey, { url: '' }, 0.05); resolve(''); }
                });
            });
            Utils.pendingSearches.set(cacheKey, promise);
            try { await promise; } finally { Utils.pendingSearches.delete(cacheKey); }
        }
        addButton(container, url) {
            if (!container || !container.isConnected) return;
            if (container.querySelector(`.fc2cmadb-search-btn[data-site="${this.name}"]`)) return;
            const btn = Utils.createButton(url, this.name, this.buttonColor);
            btn.dataset.site = this.name;
            btn.addEventListener("click", e => e.stopPropagation());
            container.appendChild(btn);
        }
    }

    // ─────────────────────────────────────────────
    // SearchManager - IntersectionObserver lazy + idle batch
    // ─────────────────────────────────────────────
    class SearchManager {
        constructor() {
            this.sites = [];
            this.activeSites = {};
            this._observer = null;
            this._lastUrl = location.href;
            this._spaTimer = null;
            this._scanScheduled = false;
            this._pendingCards = new Set();
            this._detailSeen = '';
            this._io = null;
        }
        addSite(s) { this.sites.push(s); }

        async refreshActiveSites() {
            const results = await Promise.all(this.sites.map(s => GM.getValue(`site_${s.name}`, true).catch(() => true)));
            this.sites.forEach((s, i) => this.activeSites[s.name] = results[i]);
        }

        async start() {
            await this.refreshActiveSites();
            this._setupIO();
            this.scanPage();
            this._watchDOM();
            this._watchSPA();
            // hydration fallback - 3 scans only, via idle
            let n = 0;
            const t = setInterval(() => { this.scheduleScan(); if (++n >= 3) clearInterval(t); }, 700);
        }

        _setupIO() {
            if (this._io) return;
            this._io = new IntersectionObserver((entries) => {
                for (const e of entries) {
                    if (!e.isIntersecting) continue;
                    const card = e.target;
                    this._io.unobserve(card);
                    this._processCard(card);
                }
            }, { rootMargin: '600px 0px', threshold: 0.01 });
        }

        scheduleScan() {
            if (this._scanScheduled) return;
            this._scanScheduled = true;
            const cb = () => { this._scanScheduled = false; this.scanPage(); };
            if ('requestIdleCallback' in window) requestIdleCallback(cb, { timeout: 300 });
            else setTimeout(cb, 80);
        }

        _processCard(card) {
            if (card.dataset.fc2Searched === '1') return;
            let id = null;
            // fast path: href
            const link = card.querySelector('a[href*="/articles/"]');
            if (link) {
                const href = link.getAttribute('href');
                const m = href ? href.match(RE_ARTICLE_ID) : null;
                if (m) id = m[1];
            }
            if (!id) {
                const spanEl = card.querySelector("figure span, span.absolute");
                if (spanEl) {
                    const t = spanEl.textContent.trim();
                    if (RE_NUM_5_9.test(t)) id = t;
                }
            }
            if (!id) return;
            card.dataset.fc2Searched = '1';

            let btnBox = card.querySelector(".fc2cmadb-btn-container");
            if (!btnBox) {
                btnBox = document.createElement("div");
                btnBox.className = "fc2cmadb-btn-container";
                const body = card.querySelector(".card-body") || card;
                body.appendChild(btnBox);
            }
            for (const site of this.sites) if (this.activeSites[site.name]) site.addToQueue(id, btnBox);
        }

        scanPage() {
            // 1) cards - lazy via IO if available, else immediate
            const cards = document.querySelectorAll("div.card:not([data-fc2-searched])");
            // fallback selector for edge cases
            const extra = cards.length < 8 ? document.querySelectorAll("div[class*='card']:not([data-fc2-searched]):not(.card)") : [];
            const all = extra.length ? [...cards, ...extra] : cards;
            for (const card of all) {
                if (card.dataset.fc2Searched === '1') continue;
                // if card is already in viewport or IO not supported, process immediately; else observe
                if (this._io && typeof this._io.observe === 'function') {
                    // quick viewport check to avoid IO delay for above-fold
                    const rect = card.getBoundingClientRect();
                    if (rect.top < window.innerHeight + 600) this._processCard(card);
                    else this._io.observe(card);
                } else {
                    this._processCard(card);
                }
            }

            // 2) detail page
            const m = location.pathname.match(RE_ARTICLE_ID);
            if (m) {
                const id = m[1];
                if (this._detailSeen === id && document.getElementById("detail-search-buttons")) return;
                const titleEl = document.querySelector("h1:not([data-searched])");
                if (titleEl) {
                    titleEl.dataset.searched = "1";
                    this._detailSeen = id;
                    let box = document.getElementById("detail-search-buttons");
                    if (!box) {
                        box = document.createElement("div");
                        box.id = "detail-search-buttons";
                        box.className = "fc2cmadb-btn-container";
                        box.style.marginTop = "10px";
                        box.style.marginBottom = "10px";
                        const anchorParent = titleEl.closest("a") || titleEl;
                        anchorParent.parentNode.insertBefore(box, anchorParent.nextSibling);
                        for (const site of this.sites) if (this.activeSites[site.name]) site.addToQueue(id, box);
                    }
                }
            }
        }

        _watchDOM() {
            if (this._observer) return;
            const target = document.querySelector("#app, main, [data-page], body");
            let debounce = 0;
            this._observer = new MutationObserver((mutations) => {
                // filter noise: only care if new nodes added
                let hasNew = false;
                for (const m of mutations) if (m.addedNodes.length) { hasNew = true; break; }
                if (!hasNew) return;
                clearTimeout(debounce);
                debounce = setTimeout(() => this.scheduleScan(), 180);
            });
            this._observer.observe(target, { childList: true, subtree: true });
        }

        _watchSPA() {
            const checkUrl = () => {
                if (location.href === this._lastUrl) return;
                this._lastUrl = location.href;
                // cleanup IO pending
                if (this._io) this._io.disconnect();
                const old = document.getElementById("detail-search-buttons");
                if (old) old.remove();
                const h1 = document.querySelector("h1");
                if (h1) delete h1.dataset.searched;
                this._detailSeen = '';
                clearTimeout(this._spaTimer);
                this._spaTimer = setTimeout(() => this.scanPage(), 250);
            };
            const _push = history.pushState.bind(history);
            const _replace = history.replaceState.bind(history);
            history.pushState = function (...a) { _push(...a); checkUrl(); };
            history.replaceState = function (...a) { _replace(...a); checkUrl(); };
            window.addEventListener("popstate", checkUrl);
            window.addEventListener("hashchange", checkUrl);
            document.addEventListener("inertia:finish", checkUrl);
            document.addEventListener("inertia:navigate", checkUrl);
            // also watch URL via interval for edge cases (cheap)
            setInterval(checkUrl, 1500);
        }
    }

    const manager = new SearchManager();

    manager.addSite(new Site('_123av',
        id => `https://123av.com/ja/search?keyword=${numericOnly(id)}`,
        (res, id, out) => {
            const cleanId = cleanAlnum(id);
            // finalUrl redirect check first (cheapest)
            if (res.finalUrl) {
                const cf = cleanAlnum(res.finalUrl);
                if (cf.includes(cleanId) && !res.finalUrl.includes('search')) { out.url = res.finalUrl; return; }
            }
            const doc = PARSER.parseFromString(res.responseText, "text/html");
            const link = findDirectPostLink(doc, '123av.com', id);
            if (link) { out.url = resolveAbsoluteUrl(link.getAttribute('href'), '123av.com'); return; }

                if (cf.includes(cleanId) && !res.finalUrl.includes('/search')) { out.url = res.finalUrl; return; }
            }
            const doc = PARSER.parseFromString(res.responseText, "text/html");
            // optimized single-pass scan
            const links = doc.links;
            for (let i = 0; i < Math.min(links.length, 600); i++) {
                const a = links[i];
                const href = (a.getAttribute('href') || '').toLowerCase();
                if (!href || href.includes('/search')) continue;
                const cleanHref = href.replace(RE_NON_ALNUM, '');
                if (cleanHref.includes(cleanId)) { out.url = resolveAbsoluteUrl(a.getAttribute('href'), 'missav.ws'); return; }
                const alt = (a.getAttribute('alt') || '').toLowerCase().replace(RE_NON_ALNUM, '');
                if (alt.includes(cleanId)) { out.url = resolveAbsoluteUrl(a.getAttribute('href'), 'missav.ws'); return; }
